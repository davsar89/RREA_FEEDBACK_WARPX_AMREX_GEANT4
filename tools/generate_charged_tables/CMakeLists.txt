cmake_minimum_required(VERSION 3.16)
project(RreaChargedTables LANGUAGES CXX)
find_package(Geant4 10.7.4 EXACT REQUIRED)
include(${Geant4_USE_FILE})
add_executable(rrea_charged_tables charged_tables.cc)
target_link_libraries(rrea_charged_tables PRIVATE ${Geant4_LIBRARIES})
target_compile_features(rrea_charged_tables PRIVATE cxx_std_11)
enable_testing()
add_test(NAME charged_secondary_quantiles COMMAND rrea_charged_tables --self-test)
